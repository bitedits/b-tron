Глава 1 基本Типи даних

共通して使われる型 定義 以下 記述する.

基本データ型:ITRON共通

typedef char            B;          /* 符号付き 8ビット整数 */
typedef short           H;          /* 符号付き16ビット整数 */
typedef int             W;          /* 符号付き32ビット整数 */
typedef unsigned char   UB;         /* 符号無し 8ビット整数 */
typedef unsigned short  UH;         /* 符号無し16ビット整数 */
typedef unsigned int    UW;         /* 符号無し32ビット整数 */

typedef char            VB;         /* 不定型 8ビットデータ */
typedef short           VH;         /* 不定型16ビットデータ */
typedef int             VW;         /* 不定型32ビットデータ */

typedef void            *VP;        /* 不定型データへ Вказівник (Pointer) */

typedef void            (*FP)();    /* 一般関数Вказівник (Pointer) */

以上 データ型 ITRON та 共通 ある.

Volatile: コンパイラ よる最適化禁止指定

typedef volatile W      _W;     /* 最適化禁止 符号付き 8ビット整数 */
typedef volatile H      _H;     /* 最適化禁止 符号付き16ビット整数 */
typedef volatile B      _B;     /* 最適化禁止 符号付き32ビット整数 */

typedef volatile UW     _UW;    /* 最適化禁止 符号無し 8ビット整数 */
typedef volatile UH     _UH;    /* 最適化禁止 符号無し16ビット整数 */
typedef volatile UB     _UB;    /* 最適化禁止 符号無し32ビット整数 */

typedef volatile void   _void;

Коди помилок

ビッグエンディアン 場合 

typedef union {
    W           err;            /* Коди помилок */
    struct {
            H   eclass;         /* エラークラス */
            UH  detail;         /* 詳細エラー   */
    } c;
} ErrCode;

リトルエンディアン 場合 

typedef union {
    W           err;            /* Коди помилок */
    struct {
            UH  detail;         /* 詳細エラー   */
            H   eclass;         /* エラークラス */
    } c;
} ErrCode;

Системні виклики 関数値 以下 いずれか ある. 

typedef W       ERR;        /* Коди помилок                       */
                            /* エラーまた 正常 み戻す場合 使用 */
typedef W       WERR;       /* Системні виклики関数値               */
                            /* エラーまた 意味 ある値 戻す     */
                            /* 場合 使用                         */
