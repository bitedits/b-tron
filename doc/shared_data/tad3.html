3.6 Сегменти графічного креслення та 2D-векторів (Graphic Drawing Segments)

3.6.1 Огляд векторної графіки TAD

Графічний потік TAD описує векторні геометричні примітиви (точки, відрізки ліній, прямокутники, скруглені прямокутники, еліпси, дуги, сектори, хорди, ламані лінії, багатокутники, криві Безьє/сплайни), а також патерни заповнення, колірні атрибути та графічні групи.

3.6.2 Сегменти геометричних фігур та примітивів (TS_FPRIM)

#define TS_FPRIM    0xB0    /* Геометричний примітив (лінія, прямокутник, коло тощо) */

typedef struct {
    TAD_SEG_HDR hdr;        /* id = TS_FPRIM */
    UB          prim_type;  /* Тип примітива (FP_LINE, FP_RECT, FP_OVAL, FP_POLY, FP_SPLN) */
    UB          draw_mode;  /* Режим: 0 = лише контур, 1 = заповнення, 2 = контур + заповнення */
    B           data[1];    /* Координати точок PNT / RECT / SIZE */
} TS_FPRIM_SEG;

#define FP_POINT    0x01    /* Точка */
#define FP_LINE     0x02    /* Відрізок лінії */
#define FP_RECT     0x03    /* Прямокутник */
#define FP_RRECT    0x04    /* Скруглений прямокутник */
#define FP_OVAL     0x05    /* Еліпс / Коло */
#define FP_ARC      0x06    /* Дуга */
#define FP_SECTOR   0x07    /* Сектор */
#define FP_CHORD    0x08    /* Хорда / Сегмент */
#define FP_PLINE    0x09    /* Ламана лінія (Polyline) */
#define FP_POLYGON  0x0A    /* Багатокутник (Polygon) */
#define FP_SPLINE   0x0B    /* Сплайн / Крива Безьє */

3.6.3 Сегменти визначення графічних даних (TS_FDEF)

#define TS_FDEF     0xB1    /* Визначення системного графічного об'єкта */

3.6.4 Сегменти групування графічних об'єктів (TS_FGRP)

#define TS_FGRP     0xB2    /* Початок / Завершення графічної групи (Group) */

3.6.5 Сегменти макровизначень та посилань (TS_FMAC)

#define TS_FMAC     0xB3    /* Визначення та виклик графічного макроса */

3.6.6 Сегменти графічного оформлення та патернів (TS_FATTR)

#define TS_FATTR    0xB4    /* Атрибути пера, товщини, маски ліній та патернів заливки */

typedef struct {
    TAD_SEG_HDR hdr;        /* id = TS_FATTR */
    COLOR       line_color; /* Колір контуру лінії */
    COLOR       fill_color; /* Колір заливки фігури */
    H           pen_width;  /* Товщина лінії у пікселях */
    UW          raster_op;  /* Режим накладання RasterOp (PAINT, OVER, XOR) */
} TS_FATTR_SEG;

3.6.7 Вкладка параметрів графічної сторінки (TS_FPAGE)

#define TS_FPAGE    0xC0    /* Розміри та масштаб графічного аркуша полотна */

3.6.8 Вкладка графічного коментаря (TS_FMEMO)

#define TS_FMEMO    0xC8    /* Графічна замітка / примітка */

3.6.9 Вкладка графічних даних додатків (TS_FAPPL)

#define TS_FAPPL    0xCF    /* Приватні векторні дані прикладних програм */

Додаток A: Повний перелік ідентифікаторів сегментів TAD

Диапазон ID (Hex)Призначення та категорія сегмента
0x00 .. 0x0FСистемні сегменти керування та блоків (Header, System, Text/Fig Begin/End, Image, VObj/Link)
0x10 .. 0x7FСегменти текстових вкладок (Page, Ruler, Font, Size, Align, Color, Memo, Appl)
0x80 .. 0xAFЗарезервовано для розширень текстового формату
0xB0 .. 0xCFСегменти 2D-векторної графіки (Primitives, Def, Group, Macro, Attr, Page, Memo)
0xD0 .. 0xFFЗарезервовано для системних специфічних розширень

Додаток B: Квазі-стандартні розширення TAD

Квазі-стандартні розширення визначають додаткові формати передачі стисненого звуку, відео та мультимедійних потоків усередині контейнера TAD.
