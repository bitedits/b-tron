Розділ 1. Базові типи даних та системні константи (Basic Data Types)

1.1 Вступ

У цьому розділі визначено загальні базові типи даних, специфікатори типів C, коди помилок та базові геометрії, що використовуються в усіх специфікаціях TRON / BTRON3.

1.2 Типи даних ядра та мови C

□ Базові типи даних (спільні з ITRON)

typedef char            B;          /* 8-бітне ціле зі знаком (-128..+127) */
typedef short           H;          /* 16-бітне ціле зі знаком (-32768..+32767) */
typedef int             W;          /* 32-бітне ціле зі знаком */
typedef unsigned char   UB;         /* 8-бітне ціле без знака (0..255) */
typedef unsigned short  UH;         /* 16-бітне ціле без знака (0..65535) */
typedef unsigned int    UW;         /* 32-бітне ціле без знака */

typedef char            VB;         /* 8-бітне значення неуточненого типу */
typedef short           VH;         /* 16-бітне значення неуточненого типу */
typedef int             VW;         /* 32-бітне значення неуточненого типу */

typedef void            *VP;        /* Вказівник на дані довільного типу */
typedef void            (*FP)();    /* Вказівник на універсальну функцію */

□ Специфікатор volatile (заборона оптимізації компілятора)

typedef volatile B      _B;         /* Volatile 8-бітне ціле зі знаком */
typedef volatile H      _H;         /* Volatile 16-бітне ціле зі знаком */
typedef volatile W      _W;         /* Volatile 32-бітне ціле зі знаком */

typedef volatile UB     _UB;        /* Volatile 8-бітне ціле без знака */
typedef volatile UH     _UH;        /* Volatile 16-бітне ціле без знака */
typedef volatile UW     _UW;        /* Volatile 32-бітне ціле без знака */

□ Модифікатори областей видимості та модифікаторів типів

#define CONST       const       /* Заборона модифікації константи */
#define LOCAL       static      /* Область видимості лише у межах поточного модуля (*) */
#define EXPORT                  /* Глобальний експорт символу з даного модуля */
#define IMPORT      extern      /* Зовнішній імпорт символу з іншого модуля */
#define VOID        void        /* Порожній тип */
* Модуль (Module)

Вихідний код програми, який компілюється та компонується в окремий об'єктний файл. Модуль є одиницею компіляції вихідного коду C.

□ Дійсні типи з плаваючою комою

typedef float       FLOAT;      /* Число з плаваючою комою одинарної точності (32 біти) */
typedef double      DOUBLE;     /* Число з плаваючою комою подвійної точності (64 біти) */

typedef W           (*FUNCP)(); /* Вказівник на функцію, що повертає ціле значення типу W */

□ Логічний булевий тип (Bool)

typedef enum {
    FALSE = 0,      /* Неправда (0) */
    TRUE  = 1       /* Правда (1) */
} Bool;

□ Системний ідентифікатор (ID)

typedef W       ID;         /* Системний номер/ідентифікатор об'єкта */

□ Одиниці вимірювання часу

typedef W       MSEC;       /* Час у мілісекундах */
typedef W       STIME;      /* Системний час у секундах (починаючи від 01.01.1985 00:00:00 GMT) */

□ Символьний код (TRON Code)

typedef UH      TC;         /* Символьний код TRON (16 біт на один символ) */
#define TNULL   ((TC) 0)    /* Код завершення текстового рядка (символ NUL) */

1.3 Коди помилок BTRON (System Error Codes)

Системні виклики повертають значення типу ERR або WERR: 

typedef W       ERR;        /* Код помилки (0 = успіх, < 0 = код помилки) */
typedef W       WERR;       /* Повертає статус помилки або корисне ціле значення >= 0 */
Формат структури деталей помилки (ErrCode)
/* Для архітектури Big-Endian */
typedef union {
    W   code;
    struct {
        H   eclass;     /* Клас помилки */
        UH  detail;     /* Детальний код помилки */
    } s;
} ErrCode;

/* Для архітектури Little-Endian */
typedef union {
    W   code;
    struct {
        UH  detail;     /* Детальний код помилки */
        H   eclass;     /* Клас помилки */
    } s;
} ErrCode;

□ Загальні системні константи

#define OK          (0)         /* Успішне виконання (OK) */
#define CLR         0x0000      /* Запит очищення */
#define NOCLR       0x0008      /* Запит збереження (без очищення) */
#define DELEXIT     0x8000      /* Видалення об'єкта при завершенні */

1.4 Базові геометричні типи даних

□ Розмірність (SIZE)

typedef struct {
    H   h;          /* Ширина у пікселях (горизонталь) */
    H   v;          /* Висота у пікселях (вертикаль) */
} SIZE;

□ Точка на площині (PNT)

typedef struct {
    H   x;          /* Горизонтальна координата X */
    H   y;          /* Вертикальна координата Y */
} PNT;

□ Прямокутник (RECT)

typedef struct {
    H   left;       /* Ліва координата X */
    H   top;        /* Верхня координата Y */
    H   right;      /* Права координата X (не включно) */
    H   bottom;     /* Нижня координата Y (не включно) */
} RECT;

typedef union {
    RECT    rect;
    struct {
        PNT lefttop;    /* Лівий верхній кут */
        PNT rightbot;   /* Правий нижній кут */
    } pnt;
} RECT_PNT;

□ Посилання на файл (LINK)

#define L_FSNM      20    /* Довжина назви файлової системи у символах TC */

typedef struct {
    TC    fs_name[L_FSNM];  /* Назва файлової системи */
    UH    f_id;             /* ID файлу */
    UH    atr1;             /* Атрибут 1 */
    UH    atr2;             /* Атрибут 2 */
    UH    atr3;             /* Атрибут 3 */
    UH    atr4;             /* Атрибут 4 */
    UH    atr5;             /* Атрибут 5 */
} LINK;
