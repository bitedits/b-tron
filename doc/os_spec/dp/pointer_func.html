2.6 Функції керування вказівником (Pointer Functions)

2.6.1 Вказівник миші (Pointer)

Вказівник миші (покажчик курсора) відображається поверх усіх вікон і растрових площин, автоматично слідуючи за фізичними переміщеннями координатно-вказівного пристрою (миші, тачпада чи планшета).

2.6.2 Стан та форми вказівника

  • Видимість : видимий / прихований (лічильник вимог приховування).
  • Форма та колір : стандартний системний стиль або довільний растровий бітовий патерн PTRIMG з точкою гарячого фокусу (Hot Spot).
  • Режим руху : відстеження пристрою (Tracking) або фіксована позиція (Non-tracking).

Структура образу покажчика (PTRIMG / PTRSTS)

typedef struct {
    PNT     hotpt;      /* Точка гарячого фокусу (Hot Spot) відносно лівого верхнього кута */
    B       data[72];   /* Бітова маска образу покажчика */
    B       mask[72];   /* Бітова маска прозорості/трафарету */
} PTRIMG;

typedef struct {
    W       sts;        /* Стан видимості (0 = видимий, > 0 = лічильник приховувань) */
    H       style;      /* Стиль форми (>= 0: системний, < 0: довільний PTRIMG) */
    COLOR   fgcol;      /* Колір переднього плану покажчика */
    COLOR   bgcol;      /* Колір фону/обводки покажчика */
    PNT     pos;        /* Поточна абсолютна позиція на екрані */
} PTRSTS;

2.6.3 Системні виклики керування вказівником

gini_ptr
Initialize Pointer
Ініціалізація та скидання стану вказівника

【Синтаксис / Прототип】

ERR gini_ptr(void)

【Детальний опис】

Скидає стан покажчика до стандартної стрілки зі стандартними кольорами (чорний передній план, білий фон).

gdsp_ptr
Show / Hide Pointer
Керування видимістю вказівника

【Синтаксис / Прототип】

W   gdsp_ptr(W req)

【Параметри】

W   req     0 = приховати (+1 до лічильника приховування)
            > 0 = показати (-1 до лічильника приховування)
            < 0 = примусово показати (скинути лічильник в 0)

【Значення, що повертається】

>= 0    Поточне значення лічильника приховування (0 = покажчик видимий)
gset_ptr
Set Pointer Shape and Color
Зміна форми та кольорів покажчика

【Синтаксис / Прототип】

ERR gset_ptr(PTRSTL style, PTRIMG *img, COLOR fgcol, COLOR bgcol)

【Параметри】

PTRSTL  style   Номер системного стилю (>= 0) або < 0 (використати img)
PTRIMG *img     Довільний образ покажчика
COLOR   fgcol   Передній колір (< 0 = за замовчуванням)
COLOR   bgcol   Колір фону (< 0 = за замовчуванням)
gmov_ptr
Move Pointer Position
Програмне переміщення покажчика на екрані

【Синтаксис / Прототип】

ERR gmov_ptr(PNT pos)

【Детальний опис】

Переміщує вказівник у точку pos (працює лише у режимі Non-tracking, gmot_ptr(0)).

gmot_ptr
Set Pointer Motion Tracking
Увімкнення / вимкнення відстеження фізичної миші

【Синтаксис / Прототип】

ERR gmot_ptr(W sts)

【Параметри】

W   sts     0 = зафіксувати позицію (Non-tracking); 1 = слідкувати за мишею (Tracking)
gget_ptr
Get Pointer Status
Отримання поточного стану та форми покажчика

【Синтаксис / Прототип】

W   gget_ptr(PTRSTS *sts, PTRIMG *img)

【Значення, що повертається】

>= 0    Розмір покажчика у пікселях (24x24 або 32x32)
 < 0    Помилка (EG_ADR)
