3.5 Сегменти текстових вкладок (Text Fusen Segments)

3.5.1 Огляд текстових вкладок (Text Fusen)

Текстові вкладки (Text Fusen) — це спеціальні сегменти управління, що вбудовуються безпосередньо у потік текстових символів TAD. Вони контролюють параметри макета сторінки, гарнітури шрифтів, розміри символів, відступи, лінійки абзаців, вирівнювання, табуляцію та колірне оформлення.

3.5.2 Вкладки макета та параметрів сторінки (Page Layout Fusen)

#define TS_TPAGE    0x10    /* Параметри сторінки (розмір паперу, орієнтація) */
#define TS_TPMSZ    0x11    /* Розмір друкованого поля сторінки */
#define TS_TPMMG    0x12    /* Поля сторінки (зліва, справа, зверху, знизу) */

3.5.3 Вкладки форматування рядка та абзацу (Paragraph & Ruler Fusen)

#define TS_TRULER   0x20    /* Лінійка форматування абзацу */
#define TS_TLIND    0x21    /* Лівий відступ абзацу (Left Indent) */
#define TS_TRIND    0x22    /* Правий відступ абзацу (Right Indent) */
#define TS_TFIND    0x23    /* Відступ першого рядка / абзацний відступ (First Line Indent) */
#define TS_TLSPC    0x24    /* Міжрядковий інтервал (Line Spacing) */
#define TS_TPSPC    0x25    /* Інтервал між абзацами (Paragraph Spacing) */

3.5.4 Вкладки параметрів символів та шрифту (Font & Character Specifier)

#define TS_TFONT    0x30    /* Зміна сімейства шрифту (Font Family) */
#define TS_TSIZE    0x31    /* Зміна розміру шрифту (Font Size) */
#define TS_TATTR    0x32    /* Атрибути стилю (напівжирний, курсив, підкреслення) */

typedef struct {
    TAD_SEG_HDR hdr;        /* id = TS_TFONT */
    TC          name[12];   /* Назва сімейства шрифту (TRON Code) */
} TS_TFONT_SEG;

typedef struct {
    TAD_SEG_HDR hdr;        /* id = TS_TSIZE */
    SIZE        size;       /* Розмір шрифту (ширина, висота) */
} TS_TSIZE_SEG;

3.5.5 Вкладка спеціальних символів (Symbol Fusen)

#define TS_TSYM     0x40    /* Спеціальний символ або математичний знак */

3.5.6 Вкладки вирівнювання та позиціювання (Alignment & Tabs)

#define TS_TALIGN   0x50    /* Вирівнювання тексту (зліва, по центру, справа, по ширині) */
#define TS_TTAB     0x51    /* Позиції табуляції (Tab Stops) */

#define TA_LEFT     0       /* По лівому краю */
#define TA_CENTER   1       /* По центру */
#define TA_RIGHT    2       /* По правому краю */
#define TA_JUSTIFY  3       /* По ширині (Justify) */

3.5.7 Вкладки колірного та графічного оформлення (Text Color & Style)

#define TS_TCOLOR   0x60    /* Колір тексту та колір фону під текстом */

typedef struct {
    TAD_SEG_HDR hdr;        /* id = TS_TCOLOR */
    COLOR       fgcol;      /* Колір тексту */
    COLOR       bgcol;      /* Колір фону під текстом */
} TS_TCOLOR_SEG;

3.5.8 Вкладки посилань на змінні (Variable Reference)

#define TS_TVAR     0x70    /* Динамічне поле / змінна (номер сторінки, дата, час) */

3.5.9 Вкладки текстових коментарів (Text Memo)

#define TS_TMEMO    0x78    /* Текстова замітка / примітка / коментар автора */

3.5.10 Вкладки специфічних даних додатків (Application Fusen)

#define TS_TAPPL    0x7F    /* Приватні дані прикладної програми */
